```python
import sys, re, json
rows = {}
for path in sys.argv[1:]:
    t = open(path, encoding="utf-8").read()
    i, j = t.find("{"), t.rfind("}")
    try:
        d = json.loads(t[i:j+1])
    except Exception:
        continue
    for s in d.get("sessions", []):
        c = s.get("ccr", s)
        em = c.get("external_metadata") or {}
        ps = c.get("post_turn_summary") or em.get("post_turn_summary") or {}
        rows[c["id"]] = {
            "id": c["id"], "created": c.get("created_at", ""),
            "title": c.get("title", ""), "status": c.get("session_status", ""),
            "text": " ".join(x for x in (ps.get("status_detail", ""), ps.get("recent_action", ""), ps.get("needs_action", "")) if x),
            "branches": [b for o in (c.get("session_context") or {}).get("outcomes", []) for b in ((o.get("git_repository") or {}).get("git_info") or {}).get("branches", [])],
        }
KEYS = ("noted", "unrelated", "not fixed", "not posted", "not yet", "follow", "deferred", "left",
        "could not", "couldn", "unverified", "needs", "pre-existing", "skipped", "failure", "failed",
        "uncertain", "draft", "half", "decide", "superseded", "empty", "known issue", "todo", "later")
out = ["# Session final summaries", ""]
flagged = 0
for r in sorted(rows.values(), key=lambda r: r["created"]):
    mark = "**NOTE**" if any(k in r["text"].lower() for k in KEYS) else ""
    flagged += bool(mark)
    out.append(f"- {r['created'][:16]} {r['id'][-8:]} [{r['status'][15:]}] {r['title'][:80]} {' '.join(r['branches'])} {mark}")
    if r["text"]:
        out.append("    " + r["text"][:700])
open("session-summaries.md", "w", encoding="utf-8").write("\n".join(out) + "\n")
print(len(rows), "sessions;", flagged, "flagged")
```
